-- 4. Album Information
SELECT title, artist_id 
FROM album;

-- 5. Find Bach's Tracks
SELECT name 
FROM track 
WHERE composer = 'Johann Sebastian Bach';

-- 6. Five Longest Tracks
SELECT t.name AS track_name, a.title AS album_title, t.milliseconds 
FROM track t
JOIN album a ON t.album_id = a.album_id
ORDER BY t.milliseconds DESC
LIMIT 5;

-- 7. Number of Tracks in Each Genre
SELECT g.name AS genre_name, COUNT(t.track_id) AS track_count
FROM genre g
JOIN track t ON g.genre_id = t.genre_id
GROUP BY g.name;

-- 8. Genres with Lots of Tracks
SELECT g.name AS genre_name, COUNT(t.track_id) AS track_count
FROM genre g
JOIN track t ON g.genre_id = t.genre_id
GROUP BY g.name
HAVING COUNT(t.track_id) > 30;

-- 9. Playlists and Tracks
SELECT p.name AS playlist_name, t.name AS track_name
FROM playlist p
JOIN playlist_track pt ON p.playlist_id = pt.playlist_id
JOIN track t ON pt.track_id = t.track_id;

-- 10. Artists with Most Albums
SELECT ar.artist_id, ar.name AS artist_name, COUNT(al.album_id) AS album_count
FROM artist ar
JOIN album al ON ar.artist_id = al.artist_id
GROUP BY ar.artist_id, ar.name
ORDER BY album_count DESC
LIMIT 10;
